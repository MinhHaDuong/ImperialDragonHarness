**temperatures.py**
```python
#!/usr/bin/env python3
"""Read a CSV of daily temperatures and report summary statistics."""

import argparse
import csv
import sys
from datetime import datetime


def read_temperatures(filepath):
    """Return a list of (date, temp_c) tuples from the CSV file."""
    records = []
    with open(filepath, newline="") as f:
        reader = csv.DictReader(f)
        for row in reader:
            date = datetime.strptime(row["date"], "%Y-%m-%d").date()
            temp = float(row["temp_c"])
            records.append((date, temp))
    return records


def summarize(records):
    """Return (mean, min, max, date_of_max) from a list of (date, temp) tuples."""
    if not records:
        raise ValueError("No temperature records found.")

    temps = [t for _, t in records]
    mean = sum(temps) / len(temps)
    min_temp = min(temps)
    max_temp = max(temps)
    date_of_max = max(records, key=lambda r: r[1])[0]

    return mean, min_temp, max_temp, date_of_max


def main():
    parser = argparse.ArgumentParser(description="Summarize daily temperatures from a CSV.")
    parser.add_argument("csv_file", help="Path to CSV with columns: date, temp_c")
    args = parser.parse_args()

    records = read_temperatures(args.csv_file)
    mean, min_temp, max_temp, date_of_max = summarize(records)

    print(f"Mean: {mean:.1f} °C")
    print(f"Min:  {min_temp:.1f} °C")
    print(f"Max:  {max_temp:.1f} °C on {date_of_max}")


if __name__ == "__main__":
    main()
```

**test_temperatures.py**
```python
"""Tests for temperatures.py"""

import csv
import os
import tempfile
from datetime import date

from temperatures import read_temperatures, summarize


def make_csv(rows):
    """Write rows to a temporary CSV file and return its path."""
    fd, path = tempfile.mkstemp(suffix=".csv")
    with os.fdopen(fd, "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["date", "temp_c"])
        writer.writerows(rows)
    return path


def test_summarize_basic():
    records = [
        (date(2024, 1, 1), 10.0),
        (date(2024, 1, 2), 15.0),
        (date(2024, 1, 3), 25.0),
        (date(2024, 1, 4), 5.0),
    ]
    mean, min_t, max_t, dom = summarize(records)
    assert mean == 13.75
    assert min_t == 5.0
    assert max_t == 25.0
    assert dom == date(2024, 1, 3)


def test_summarize_single():
    records = [(date(2024, 6, 1), 30.0)]
    mean, min_t, max_t, dom = summarize(records)
    assert mean == 30.0
    assert min_t == 30.0
    assert max_t == 30.0
    assert dom == date(2024, 6, 1)


def test_read_temperatures():
    path = make_csv([
        ["2024-03-01", "8.5"],
        ["2024-03-02", "12.0"],
    ])
    try:
        records = read_temperatures(path)
        assert len(records) == 2
        assert records[0] == (date(2024, 3, 1), 8.5)
        assert records[1] == (date(2024, 3, 2), 12.0)
    finally:
        os.unlink(path)


def test_end_to_end():
    path = make_csv([
        ["2024-07-01", "20.0"],
        ["2024-07-02", "35.5"],
        ["2024-07-03", "18.0"],
    ])
    try:
        records = read_temperatures(path)
        mean, min_t, max_t, dom = summarize(records)
        assert mean == 24.5
        assert min_t == 18.0
        assert max_t == 35.5
        assert dom == date(2024, 7, 2)
    finally:
        os.unlink(path)
```
